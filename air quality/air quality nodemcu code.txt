#include <ESP8266WiFi.h>
#include <ESP8266HTTPClient.h>
#include <WiFiClientSecure.h>
#include <DHT.h>



#define DHTPIN D1
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);


// ================= WIFI =================

#define WIFI_SSID "Wifi Name"
#define WIFI_PASSWORD "Wifi Password"



// ================= SPRING BOOT IP =================

#define HOST "System IP address"



// ================= TELEGRAM =================

String botToken = "Token ID";

String chatID = "ChatID";



// ================= DHT11 =================


int humidity = 0;
int temperature = 0;



void setup()
{

  Serial.begin(115200);

  delay(1000);

  Serial.println("System Starting...");


  // DHT11
  dht.begin();


  // MQ135
  pinMode(A0, INPUT);


  // ================= WIFI =================

  WiFi.mode(WIFI_STA);

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  Serial.print("Connecting to WiFi");

  while (WiFi.status() != WL_CONNECTED)
  {
    delay(500);

    Serial.print(".");
  }

  Serial.println("");

  Serial.println("WiFi Connected");

  Serial.print("IP Address : ");

  Serial.println(WiFi.localIP());

}



void loop()
{


// WIFI RECONNECT

if(WiFi.status() != WL_CONNECTED)
{
    Serial.println("WiFi Disconnected");

    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

    delay(5000);

    return;
}


  

  // ================= DHT11 =================

  temperature = dht.readTemperature();

  humidity = dht.readHumidity();


  // INVALID SENSOR VALUES FIX

  if (isnan(temperature))
  {
    temperature = -1;
  }

  if (isnan(humidity))
  {
    humidity = -1;
  }




// ================= SAFE DHT11 READING ================= 
float tempRead = dht.readTemperature(); 
float humRead = dht.readHumidity(); 

// DHT11 DISCONNECTED 
if(isnan(tempRead) || isnan(humRead)) {
  temperature = -1;
  humidity = -1; 
  } 
  else {
    temperature = (int)tempRead; 
    humidity = (int)humRead; 
    }
  



  // ================= MQ135 =================

// ================= SAFE MQ135 READING ================= 
int mq135;

int mqRead = analogRead(A0); 

// MQ135 DISCONNECTED 
if(mqRead <= 5) { 
  mq135 = -1; 
  } 
  else { 
    mq135 = mqRead; 
    }



  // ================= SERIAL MONITOR =================

  Serial.println("========== SENSOR DATA ==========");

  Serial.print("Temperature : ");

  Serial.println(temperature);

  Serial.print("Humidity : ");

  Serial.println(humidity);

  Serial.print("Gas Value : ");

  Serial.println(mq135);

  Serial.println("=================================");



// ================= TELEGRAM ALERT =================

static bool telegramSent = false;

// DANGER LEVEL
if(mq135 > 100)   // Change threshold as needed
{
    if(!telegramSent)
    {
        WiFiClientSecure client;
        client.setInsecure();

        HTTPClient http;

        String url =
        "https://api.telegram.org/bot" +
        botToken +
        "/sendMessage?chat_id=" +
        chatID +
        "&text=AIR_QUALITY_DANGER!_AQI_" +
        String(mq135) +
        "_TEMP_" +
        String(temperature) +
        "_HUM_" +
        String(humidity);

        http.begin(client, url);

        int code = http.GET();

        Serial.print("Telegram Code: ");
        Serial.println(code);

        Serial.println(http.getString());

        http.end();

        telegramSent = true;
    }
}
else
{
    telegramSent = false;
}

  // ================= SEND DATA TO SPRING =================

  WiFiClient client;

  HTTPClient http;


  String temp1 = String(temperature);

  String hum1 = String(humidity);

  String gas = String(mq135);



  String url =
    "http://" + String(HOST) +
    ":9094/weatherapp/" +
    temp1 + "/" +
    hum1 + "/" +
    gas;


  Serial.println(url);



  http.begin(client, url);

  int httpCode = http.GET();



  Serial.print("HTTP Response : ");

  Serial.println(httpCode);



  if (httpCode == 200)
  {

    Serial.println("Values uploaded successfully");

  }

  else
  {

    Serial.println("Upload Failed");

  }



  http.end();


  delay(1000);

}
